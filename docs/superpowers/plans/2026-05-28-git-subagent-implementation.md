# Git Subagent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Activate the git subagent with full read+write capability, wire it into the root pipeline as a planner-driven post-editor step, and expose git read tools to the reader subagent.

**Architecture:** Approach C — `filesToStage` is derived deterministically from `editorAppliedFiles` (surfaced by fixing `delegateWriterNode`); commit message is generated by one structured LLM call in `buildGitIntentNode`; `executeGitOps` runs branch→add→commit/amend→push-gate sequentially in one node; push always requires user approval via `interrupt()`.

**Tech Stack:** LangGraph (`StateGraph`, `interrupt`), Zod, LangChain tools (`tool()`), `runGit` from `@robocode-packages/shared`, `EventBus` from `@robocode-packages/core`.

```typescript
import { tool } from '@langchain/core/tools';
import { z } from 'zod';
import { TOOL_NAMES, runGit } from '@robocode-packages/shared';
import type { RunnableConfig } from '@langchain/core/runnables';

export const gitAmendTool = tool(
  async ({ files, message }, config?: RunnableConfig) => {
    const cwd = (config?.configurable?.cwd as string) ?? process.cwd();

    // Warn if branch has a remote tracking ref (amending a pushed commit needs force push)
    const remoteCheck = await runGit('log --remotes --oneline -1', cwd);
    const hasRemote = remoteCheck !== '(no output)' && !remoteCheck.startsWith('Error:');
    const warning = hasRemote
      ? 'Warning: branch appears to have remote commits — amending will diverge from remote.\n'
      : '';

    const fileArgs = files.map((f) => `"${f}"`).join(' ');
    const addResult = await runGit(`add ${fileArgs}`, cwd);
    if (addResult.startsWith('Error:')) return `${warning}${addResult}`;

    const messageFlag = message
      ? `-m "${message.replace(/"/g, '\\"')}"`
      : '--no-edit';
    const amendResult = await runGit(`commit --amend ${messageFlag}`, cwd);
    return `${warning}${amendResult}`;
  },
  {
    name: TOOL_NAMES.git_amend,
    description: 'Stage files and amend the last commit. Optionally provide a new message; omit to keep existing message.',
    schema: z.object({
      files: z.array(z.string()).min(1).describe('Files to stage before amending'),
      message: z.string().optional().describe('New commit message. Omit to keep existing message (--no-edit).'),
    }),
  }
);
```

```typescript
import { tool } from '@langchain/core/tools';
import { z } from 'zod';
import { TOOL_NAMES, runGit } from '@robocode-packages/shared';
import type { RunnableConfig } from '@langchain/core/runnables';

export const gitCheckoutBranchTool = tool(
  async ({ branch }, config?: RunnableConfig) => {
    const cwd = (config?.configurable?.cwd as string) ?? process.cwd();
    const result = await runGit(`checkout -b "${branch}"`, cwd);
    if (result.startsWith('Error:') && result.includes('already exists')) {
      return `Branch "${branch}" already exists. Switching to it.`;
    }
    return result;
  },
  {
    name: TOOL_NAMES.git_checkout_branch,
    description: 'Create and switch to a new branch. If the branch already exists, switches to it without error.',
    schema: z.object({
      branch: z.string().min(1).describe('Branch name to create, e.g. "feature/add-faq-route"'),
    }),
```

```typescript
import { tool } from '@langchain/core/tools';
import { z } from 'zod';
import { TOOL_NAMES, runGit } from '@robocode-packages/shared';
import type { RunnableConfig } from '@langchain/core/runnables';

export const gitStashTool = tool(
  async ({ action, message }, config?: RunnableConfig) => {
    const cwd = (config?.configurable?.cwd as string) ?? process.cwd();
    if (action === 'list') return runGit('stash list', cwd);
    if (action === 'pop') return runGit('stash pop', cwd);
    const msgFlag = message ? ` -m "${message.replace(/"/g, '\\"')}"` : '';
    return runGit(`stash push${msgFlag}`, cwd);
  },
  {
    name: TOOL_NAMES.git_stash,
    description: 'Stash, pop, or list stashes. Use push to save uncommitted changes, pop to restore them.',
    schema: z.object({
      action: z.enum(['push', 'pop', 'list']).describe('Stash operation'),
      message: z.string().optional().describe('Description for stash push'),
    }),
  }
);
```

```typescript
import { debug } from '@robocode-packages/shared';
import { runGit } from '@robocode-packages/shared';
import type { GitStateType } from '../../../main/subagents/git/state';
import type { GitOpResult } from '@robocode-packages/shared';
import type { PendingToolCall } from '@robocode-packages/shared';
import { TOOL_NAMES } from '@robocode-packages/shared';

export const executeGitOpsNode = async (
  state: GitStateType
): Promise<{
  gitOpResults: GitOpResult[];
  pendingToolCall: PendingToolCall | null;
}> => {
  const { gitIntent, cwd } = state;
  const results: GitOpResult[] = [];

  if (!gitIntent) {
    debug('[executeGitOps] no gitIntent — skipping');
    return { gitOpResults: [], pendingToolCall: null };
  }

  const { branch, filesToStage, commitMessage, push, amend } = gitIntent;

  // Step 1: checkout branch (conditional)
  if (branch) {
    debug('[executeGitOps] creating branch:', branch);
    const out = await runGit(`checkout -b "${branch}"`, cwd);
    const failed = out.startsWith('Error:') && !out.includes('already exists');
    results.push({ op: 'checkout_branch', status: failed ? 'failed' : 'ok', output: out });
  }

  // Step 2: stage files
  if (filesToStage.length === 0) {
    results.push({ op: 'add', status: 'skipped', output: 'no files to stage' });
  } else {
    debug('[executeGitOps] staging:', filesToStage);
    const fileArgs = filesToStage.map((f) => `"${f}"`).join(' ');
    const addOut = await runGit(`add ${fileArgs}`, cwd);
    results.push({ op: 'add', status: addOut.startsWith('Error:') ? 'failed' : 'ok', output: addOut });
  }

  // Step 3: check if anything is staged
  const diffIndex = await runGit('diff --cached --name-only', cwd);
  const hasStaged = diffIndex !== '(no output)' && !diffIndex.startsWith('Error:') && diffIndex.trim() !== '';

  if (!hasStaged) {
    debug('[executeGitOps] nothing staged — skipping commit');
    results.push({ op: amend ? 'amend' : 'commit', status: 'skipped', output: 'nothing staged' });
    return { gitOpResults: results, pendingToolCall: null };
  }

  // Step 4: commit or amend
  let commitOut: string;
  if (amend) {
    debug('[executeGitOps] amending last commit');
    const msgFlag = `-m "${commitMessage.replace(/"/g, '\\"')}"`;
    commitOut = await runGit(`commit --amend ${msgFlag}`, cwd);
    results.push({ op: 'amend', status: commitOut.startsWith('Error:') ? 'failed' : 'ok', output: commitOut });
  } else {
    debug('[executeGitOps] committing:', commitMessage);
    commitOut = await runGit(`commit -m "${commitMessage.replace(/"/g, '\\"')}"`, cwd);
    results.push({ op: 'commit', status: commitOut.startsWith('Error:') ? 'failed' : 'ok', output: commitOut });
  }

  // Step 5: push gate
  if (!push) {
    return { gitOpResults: results, pendingToolCall: null };
  }

  const pendingToolCall: PendingToolCall = {
    id: crypto.randomUUID(),
    name: TOOL_NAMES.git_push,
    args: {},
    risk: 'destructive',
    description: `Push branch "${branch ?? 'current'}" to origin\nCommit: ${commitMessage}`,
  };

  debug('[executeGitOps] push requested — awaiting approval');
  return { gitOpResults: results, pendingToolCall };
};

export const afterExecuteGitOps = (state: GitStateType): 'push_approval' | 'final' =>
  state.pendingToolCall ? 'push_approval' : 'final';
```

```typescript
import { debug } from '@robocode-packages/shared';
import { writerAgent } from '../../main/subagents/editor';
import type { RootStateType } from '../../main/root';
import type { IntentSchemaType } from '@robocode-packages/shared';

type EditResult = { file: string; status: string };

export const extractAppliedFiles = (
  editIntent: IntentSchemaType | null,
  editResults: EditResult[]
): string[] => {
  if (editResults.length > 0) {
    return editResults
      .filter((r) => r.status === 'applied' || r.status === 'recovered')
      .map((r) => r.file);
  }
  // Fallback: editor was interrupted before completing — use planned files
  return editIntent?.edits.map((e) => e.file) ?? [];
};

export const delegateWriterNode = async (state: RootStateType) => {
  const { editIntent, sessionId, cwd, autoApprove } = state;

  if (!editIntent) {
    debug('[delegateWriter] no editIntent — skipping');
    return {};
  }

  debug('[delegateWriter] starting editor with', editIntent.edits.length, 'edits');

  await writerAgent.deleteCheckpoint();

  try {
    const result = await writerAgent.run({
      editIntent,
      sessionId,
      cwd,
      autoApprove: autoApprove ?? false,
    });
    const editorAppliedFiles = extractAppliedFiles(editIntent, result.editResults ?? []);
    debug('[delegateWriter] applied files:', editorAppliedFiles);
    return { editorAppliedFiles };
  } catch (err: unknown) {
    const isInterrupt =
      err != null &&
      typeof err === 'object' &&
      'name' in err &&
      (err as { name: string }).name === 'GraphInterrupt';
    if (!isInterrupt) throw err;
    const editorAppliedFiles = extractAppliedFiles(editIntent, []);
    debug('[delegateWriter] editor interrupted — using editIntent files as fallback:', editorAppliedFiles);
    return { editorAppliedFiles };
  }
};
```

Create `packages/agent/src/nodes/root/buildGitIntent.ts`:

```typescript
import { HumanMessage, SystemMessage } from '@langchain/core/messages';
import { z } from 'zod';
import { debug } from '@robocode-packages/shared';
import { getModel } from '../utils';
import type { RootStateType } from '../../main/root/state';
import type { GitIntentType } from '@robocode-packages/shared';

const CommitMessageSchema = z.object({
  message: z.string().describe('Conventional commit message, one line, under 72 chars'),
});

const COMMIT_MESSAGE_SYSTEM = `You write conventional git commit messages.
Format: type(scope): short description
Types: feat | fix | refactor | chore | docs | test | style
One line only. Under 72 characters. No body, no footer.`;

const buildHumanPrompt = (goal: string, files: string[], edits: Array<{ mode: string; action: string; file: string }>) => {
  const editLines = edits.map((e) => `- ${e.mode}/${e.action} ${e.file}`).join('\n');
  return `Goal: ${goal}\nFiles changed:\n${files.map((f) => `- ${f}`).join('\n')}\nChanges:\n${editLines || '(no detail available)'}`;
};

export const buildGitIntentNode = async (state: RootStateType): Promise<{ gitIntent: GitIntentType | null }> => {
  const { plan, editorAppliedFiles, editIntent } = state;

  if (!plan?.gitStep) {
    debug('[buildGitIntent] no gitStep in plan — skipping');
    return { gitIntent: null };
  }

  const filesToStage = [...new Set(editorAppliedFiles)];
  const { branch, push = false, amend = false } = plan.gitStep;

  debug('[buildGitIntent] generating commit message for', filesToStage.length, 'files');

  const model = getModel(false).withStructuredOutput(CommitMessageSchema, { name: 'commit_message' });
  const edits = (editIntent?.edits ?? []).map((e) => ({ mode: e.mode, action: e.action, file: e.file }));
  const { message: commitMessage } = await model.invoke([
    new SystemMessage(COMMIT_MESSAGE_SYSTEM),
    new HumanMessage(buildHumanPrompt(plan.goal, filesToStage, edits)),
  ]);

  const gitIntent: GitIntentType = { filesToStage, commitMessage, push, amend, ...(branch ? { branch } : {}) };
  debug('[buildGitIntent] gitIntent:', gitIntent);
  return { gitIntent };
};
```

```typescript
import { debug } from '@robocode-packages/shared';
import { gitAgent } from '../../main/subagents/git';
import type { RootStateType } from '../../main/root';

export const delegateGitNode = async (state: RootStateType) => {
  const { gitIntent, sessionId, cwd } = state;

  if (!gitIntent) {
    debug('[delegateGit] no gitIntent — skipping');
    return {};
  }

  debug('[delegateGit] running git subagent');

  try {
    await gitAgent.run({
      task: `Git operations: ${gitIntent.commitMessage}`,
      cwd: cwd ?? process.cwd(),
      gitIntent,
    });
  } catch (err: unknown) {
    const isInterrupt =
      err != null &&
      typeof err === 'object' &&
      'name' in err &&
      (err as { name: string }).name === 'GraphInterrupt';
    if (!isInterrupt) throw err;
    debug('[delegateGit] git subagent interrupted — awaiting push approval');
  }

  return {};
};
```
